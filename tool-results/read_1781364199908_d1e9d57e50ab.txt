    57→// Undo/Redo history entry
    58→interface UndoEntry {
    59→  scenarioId: string
    60→  apiField: EditableField
    61→  oldValue: string
    62→  newValue: string
    63→  clientId?: string // for company edits

    66→function getColValue(scenario: any, key: ColFilterKey): string {
    67→  switch (key) {
    68→    case 'name': return scenario.name || ''
    69→    case 'status': return scenario.status || ''
    70→    case 'company': return scenario.company || scenario.client?.company || scenario.client?.name || ''
    71→    case 'website': return scenario.companyWebsiteUrl || ''
    72→    case 'overview': return scenario.overview || ''
    73→    case 'updatedAt': return scenario.updatedAt ? new Date(scenario.updatedAt).toLocaleDateString() : ''
    74→    case 'createdAt': return scenario.createdAt ? new Date(scenario.createdAt).toLocaleDateString() : ''
    75→    default: return ''
    76→  }
    77→}
    78→
    79→// Get raw value for editing from a scenario object
    80→function getEditValue(scenario: any, apiField: EditableField): string {
    81→  if (apiField === 'companyWebsiteUrl') return scenario.companyWebsiteUrl || ''
    82→  if (apiField === 'company') return scenario.company || scenario.client?.company || ''
    83→  return (scenario as any)[apiField] || ''
    84→}
